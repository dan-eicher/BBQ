cmake_minimum_required(VERSION 3.16)
project(pegc LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# --- pegc frontend library ---
add_library(pegc_frontend STATIC
    frontend/Parser.cpp
    frontend/PegAnalysis.cpp
)
target_include_directories(pegc_frontend PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/frontend
    ${CMAKE_CURRENT_SOURCE_DIR}/runtime
    ${CMAKE_SOURCE_DIR}/lib)

# --- pegc backend library ---
# Reuses FrameProcessor from BBQ frontend
add_library(pegc_backend STATIC
    backend/ParserEmitter.cpp
    backend/CParserEmitter.cpp
    backend/JavaParserEmitter.cpp
    runtime/cpp_emit.cpp
    runtime/c_emit.cpp
    runtime/java_emit.cpp
    ../frontend/FrameProcessor.cpp
)
target_include_directories(pegc_backend PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/backend
    ${CMAKE_CURRENT_SOURCE_DIR}/runtime)
target_include_directories(pegc_backend PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/../frontend)
target_link_libraries(pegc_backend PUBLIC pegc_frontend)

# --- pegc code generator ---
add_executable(pegc tools/pegc.cpp)
target_link_libraries(pegc PRIVATE pegc_backend)

# Self-bootstrap regeneration. Out of the default build graph because
# pegc -> pegc_frontend -> Parser.cpp would form a cycle if Parser.cpp
# depended on pegc. Same story for GrammarAST.h, which is generated by
# asdl, which depends on pegc. Both are checked-in artifacts; rebuild
# explicitly when changing peg.peg or grammar.asdl:
#   cmake --build build --target regen_pegc_parser
#   cmake --build build --target regen_pegc_ast
# After regen, a normal `cmake --build` picks up the new files.
add_custom_target(regen_pegc_parser
    COMMAND pegc ${CMAKE_CURRENT_SOURCE_DIR}/grammar/peg.peg
                 -frames ${CMAKE_CURRENT_SOURCE_DIR}/grammar/frames/
                 -o ${CMAKE_CURRENT_SOURCE_DIR}/frontend/
                 -prefix ""
    DEPENDS pegc
    COMMENT "Regenerating pegc Parser.h/.cpp from peg.peg"
    VERBATIM
)

add_custom_target(regen_pegc_ast
    COMMAND asdl -i ${CMAKE_CURRENT_SOURCE_DIR}/grammar/grammar.asdl
                 -t ${CMAKE_CURRENT_SOURCE_DIR}/grammar/templates/grammar_ast.inja
                 -o ${CMAKE_CURRENT_SOURCE_DIR}/frontend/GrammarAST.h
    DEPENDS asdl
    COMMENT "Regenerating pegc GrammarAST.h from grammar.asdl"
    VERBATIM
)

# ParserKont is a checked-in artifact for the same reason peg.peg's
# Parser.cpp is: asdl depends on pegc, pegc depends on pegc_backend,
# any consumer in pegc_backend would close the loop. Run explicitly:
#   cmake --build build --target regen_parser_kont
add_custom_target(regen_parser_kont
    COMMAND asdl -i ${CMAKE_CURRENT_SOURCE_DIR}/grammar/ParserKont.asdl
                 -t ${CMAKE_CURRENT_SOURCE_DIR}/grammar/templates/grammar_ast.inja
                 -o ${CMAKE_CURRENT_SOURCE_DIR}/runtime/ParserKont.h
    COMMAND asdl -i ${CMAKE_CURRENT_SOURCE_DIR}/grammar/ParserKont.asdl
                 --json ${CMAKE_CURRENT_SOURCE_DIR}/grammar/ParserKont.json
    DEPENDS asdl
    COMMENT "Regenerating ParserKont.h + ParserKont.json from ParserKont.asdl"
    VERBATIM
)

# --- tests ---
find_package(GTest REQUIRED)
enable_testing()

add_executable(pegc_tests
    tests/test_grammar_parser.cpp
    tests/test_runtime.cpp
    tests/test_peg_analysis.cpp
    tests/test_peg_analysis_kildall.cpp
    tests/test_codegen.cpp
    tests/test_java_codegen.cpp
)
target_link_libraries(pegc_tests PRIVATE pegc_backend GTest::GTest GTest::Main)
target_compile_definitions(pegc_tests PRIVATE
    SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME pegc_tests COMMAND pegc_tests)

# C backend e2e: the halt contract. The only other consumer of -lang c
# output is outside this tree, so the contract is pinned here.
set(PEGC_HALT_OUT ${CMAKE_CURRENT_BINARY_DIR}/halt_gen)
add_custom_command(
    OUTPUT ${PEGC_HALT_OUT}/halt_parser.h ${PEGC_HALT_OUT}/halt_parser.c
    COMMAND ${CMAKE_COMMAND} -E make_directory ${PEGC_HALT_OUT}
    COMMAND pegc ${CMAKE_CURRENT_SOURCE_DIR}/tests/data/halt.peg -lang c
                 -frames ${CMAKE_CURRENT_SOURCE_DIR}/backend/frames
                 -o ${PEGC_HALT_OUT}/
    DEPENDS pegc ${CMAKE_CURRENT_SOURCE_DIR}/tests/data/halt.peg
            ${CMAKE_CURRENT_SOURCE_DIR}/backend/frames/CParserImpl.frame
    COMMENT "Running pegc -lang c on tests/data/halt.peg"
    VERBATIM)
add_executable(pegc_halt_c_e2e
    tests/data/halt_main.c
    ${PEGC_HALT_OUT}/halt_parser.c)
set_target_properties(pegc_halt_c_e2e PROPERTIES
    LINKER_LANGUAGE C C_STANDARD 11 C_STANDARD_REQUIRED ON)
target_compile_options(pegc_halt_c_e2e PRIVATE -Wall -Wextra -Werror)
target_include_directories(pegc_halt_c_e2e PRIVATE
    ${PEGC_HALT_OUT} ${CMAKE_CURRENT_SOURCE_DIR}/runtime)
add_test(NAME pegc_halt_c_e2e COMMAND pegc_halt_c_e2e)
